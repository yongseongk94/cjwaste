function pointSegmentDistanceKm(lat,lng,a,b){
  if(!a||!b)return Infinity;
  const p=geoPointKm(lat,lng), p1=geoPointKm(a.lat,a.lng), p2=geoPointKm(b.lat,b.lng);
  const vx=p2.x-p1.x, vy=p2.y-p1.y;
  const wx=p.x-p1.x, wy=p.y-p1.y;
  const vv=vx*vx+vy*vy;
  let t=vv>0?(wx*vx+wy*vy)/vv:0;
  t=Math.max(0,Math.min(1,t));
  return Math.hypot(p.x-(p1.x+t*vx),p.y-(p1.y+t*vy));
}

function pointSegmentDistanceKm(lat,lng,a,b){
  if(!a||!b)return Infinity;
  const p=geoPointKm(lat,lng), p1=geoPointKm(a.lat,a.lng), p2=geoPointKm(b.lat,b.lng);
  const vx=p2.x-p1.x, vy=p2.y-p1.y;
  const wx=p.x-p1.x, wy=p.y-p1.y;
  const vv=vx*vx+vy*vy;
  let t=vv>0?(wx*vx+wy*vy)/vv:0;
  t=Math.max(0,Math.min(1,t));
  return Math.hypot(p.x-(p1.x+t*vx),p.y-(p1.y+t*vy));
}
